Excluding participants whose accuracy is too close to chance level

Here, we check the participants’ accuracy across all trials. If the accuracy rate is at chance level, the participants are excluded from further data analysis, as it is too probable that they have just answered randomly.

# making a "correct/incorrect"-column
df_clean$correct_incorrect <- ifelse(
  df_clean$Ychange_Nchange_response == "no" & df_clean$correct_change_answer == "no_change" |
    df_clean$Ychange_Nchange_response == "yes" & df_clean$correct_change_answer == "affix_deletion" & df_clean$response == df_clean$correct_response |
    df_clean$Ychange_Nchange_response == "yes" & df_clean$correct_change_answer == "other_change" & df_clean$response == df_clean$correct_response |
    df_clean$Ychange_Nchange_response == "yes" & df_clean$correct_change_answer == "subject_deletion"& df_clean$response == df_clean$correct_response,
  1, 0)

# how many correct does each participant have?
df_correct_count <- df_clean %>% 
  group_by(participant) %>% 
  summarize(correct_count = sum(correct_incorrect))

# looking at the binomial distribution, we can see that the cutoff point must be 43 incorrect answers (p = 0.05).

Discarding individual outlier trials with very long reaction times

We discard trials where the sum of the reaction times on the three stages (initial sentence reading, change detection, change localization) is more than three SDs above the participant’s own mean. We do this because very long RTs can indicate that the participant zoned out or did not pay attention to the task.

# get reaction times for all trials
rt_different_stages_full <- merge(merge(reading_time_first_sentence, change_detection_time), localization_time) %>%
  filter(!(sentence_id_specific %in% c("p1", "p2", "p3", "p4", NA))) %>%
  mutate(sum_rt = reading_time_sentence1 + change_detection_time + localization_time)

# calculate participant means and SDs - and merge 
rt_mean_and_SD <- rt_different_stages_full %>%
  group_by(participant) %>%
  summarise(mean_rt = mean(sum_rt),
            sd_rt = sd(sum_rt))
rt_different_stages_full_merged <- merge(rt_different_stages_full, rt_mean_and_SD, by = "participant")

# get outlier trials
rt_outliers <- rt_different_stages_full_merged %>%
  filter(sum_rt > mean_rt + 3 * sd_rt)

# exclude outliers
df_clean <- df_clean %>%
  anti_join(rt_outliers, by = c("participant", "sentence_id_specific"))

But what if we standardize the reading time measure within each participant? Then we can see if reading faster or slower than one usually does impacts change detection accuracy for affixes.

# first calculate RT pr character for sentence 1 in each trial
# (because the sentences are not all equal in length)

sentence_df <- df_clean %>%
  group_by(sentence_id_specific) %>%
  summarise(sentence1 = first(original_sentence))

rt_per_character <- reading_time_first_sentence %>%
  filter(!(sentence_id_specific %in% c("p1", "p2", "p3", "p4", NA)))

rt_per_character$sentence1 <- sentence_df$sentence1[
  match(rt_per_character$sentence_id_specific, sentence_df$sentence_id_specific)
]

rt_per_character <- rt_per_character %>%
  mutate(rt_per_char = reading_time_sentence1 / nchar(sentence1))

# then center and standardize this measure within each participant

rt_per_character <- rt_per_character %>%
